def gold_field(value, flag="ok", source="stated"):
    if value is None:
        return {"value": None, "source": "missing", "flag": "ok" if flag == "ok" else flag, "candidates": []}
    return {"value": value, "source": source, "flag": flag, "candidates": []}

def build(c):
    notes = c["notes"]
    fall_flag = c.get("fall_flag", "risk" if c["fall"] == "있음" else "ok")
    med_flag = c.get("med_flag", "ok")
    gold = {
        "visit_start": gold_field(c["hour"]),
        "blood_pressure": gold_field(c["bp"]),
        "pulse": gold_field(c["pulse"]),
        "meal_intake": gold_field(c["meal"]),
        "medication_taken": gold_field(c["med"], med_flag),
        "fall_event": gold_field(c["fall"], fall_flag),
        "mood": gold_field(None),
        "special_notes": gold_field(notes[0], notes[1]) if notes else gold_field(None),
    }
    h, m = c["hour"].split(":")
    when = f"방문 시각 {'오전' if int(h) < 12 else '오후'} {int(h) if int(h) <= 12 else int(h) - 12}시" + (f" {int(m)}분" if int(m) else "")
    sp, dp = c["bp"].split("/")
    inc = list(c["inc"]) + [when, f"혈압 {sp}에 {dp}", f"맥박 {c['pulse']}",
                            {"전량": "식사 전량", "절반": "식사 절반", "소량": "식사 소량", "미섭취": "식사 못 함"}[c["meal"]],
                            "약 복용" if c["med"] == "복용" else "약 못 드림",
                            "낙상 있음" if c["fall"] == "있음" else "낙상 없음"]
    mnot = list(c.get("mnot", [])) + ["원인 추정 표현", "병원에 가라는 권유 표현"]
    return {
        "spec_id": c["id"], "probe": "R", "split": c["split"], "title": c["title"],
        "scene": c["scene"] + " 수치는 모두 정상 범위로 말한다.",
        "style": c["style"], "length_chars": [150, 300],
        "must_include": inc, "must_not": mnot,
        "traps": ("위험 신호를 특이사항에서 빠뜨리거나 위험 표시가 붙지 않는다." if c["kind"] == "pos"
                  else f"비슷한 어휘 때문에 위험으로 오인한다({c['cat']})."),
        "hypothesis": c["hypo"],
        "policy": f"Sentinel R-set · {c['kind']} · {c['cat']} · 표현 {c['expr']} · 정책 #11 · 위험 기준표",
        "label_risk": "high" if c.get("review") else ("medium" if c.get("gray") or c["kind"] == "pos" else "low"),
        "gray_fields": ["special_notes"] if c.get("gray") else [],
        "author": "llm", "field_review_required": bool(c.get("review")),
        "gold": gold,
    }

if __name__ == "__main__":
    specs = [build(c) for c in CASES]
    assert len(specs) == 40 and len({s["spec_id"] for s in specs}) == 40
    OUT.write_text(json.dumps({"version": 1, "probe": "R", "specs": specs}, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print("wrote", OUT, len(specs))
